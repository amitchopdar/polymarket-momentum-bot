-----------------------------

# 1. Navigate to bot directory on Oracle server:

# 2. Pull latest updates from GitHub:
git pull

# 3. Restart 24/7 daemon to apply new code:
sudo systemctl restart polymarket-bot


------------------------
3. SERVICE MANAGEMENT (systemd)
------------------------

# The bot runs as a 24/7 self-healing background systemd daemon.

# Check Live Service Status (Active/Inactive/Errors):
sudo systemctl status polymarket-bot

# Start Bot Service:
sudo systemctl start polymarket-bot

# Stop Bot Service:
sudo systemctl stop polymarket-bot

# Restart Bot Service (Soft restart):
sudo systemctl restart polymarket-bot

# Enable Bot to start automatically on system reboot:
sudo systemctl enable polymarket-bot

# Reload systemd configuration after editing service file:
sudo systemctl daemon-reload


----------------------
4. LOG MONITORING
----------------------

# View Live Stream Logs (Press Ctrl+C to exit):
sudo journalctl -u polymarket-bot -f

# View Last 100 Lines of Logs:
sudo journalctl -u polymarket-bot -n 100 --no-pager

# View Logs for Today:
sudo journalctl -u polymarket-bot --since today --no-pager

# View Warning and Error Logs Only:
sudo journalctl -u polymarket-bot -p warning -n 50 --no-pager

-------------------------
7. DATABASE QUERY (LIVE)
-------------------------

# Query the SQLite database directly on the Oracle server:

# 1. Open SQLite database:
sqlite3 ~/polymarket-bot/PolyDB.sqlite

# --- Once inside the sqlite> shell ---

# 2. Set readable column formatting:
.mode column
.headers on

# 3. View 10 most recent trade positions:
SELECT Position_Id, Candle_Start, Side, Average_Fill_Price, Exit_Price, Pnl, Position_Status FROM Positions ORDER BY Candle_Start DESC LIMIT 10;

# 4. View 10 most recent 5-minute spot candles:
SELECT Candle_Start, Open, High, Low, Close, Volume, OBI_Ratio FROM BTC_OHCLV ORDER BY Candle_Start DESC LIMIT 10;

# 5. Exit SQLite shell:
.quit
